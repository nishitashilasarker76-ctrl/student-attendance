📄 sketch.ino — Main Arduino Code

// ============================================ // Sensor-Based Attendance System — Wokwi Simulation // ESP32 + PIR + RFID + Ultrasonic + OLED + Buzzer // ============================================ #include <SPI.h> #include <MFRC522.h> #include <Wire.h> #include <Adafruit_GFX.h> #include <Adafruit_SSD1306.h> #include <WiFi.h> // ---- Pin Definitions ---- #define PIR_PIN 27 // PIR motion sensor #define TRIG_PIN 13 // Ultrasonic trigger #define ECHO_PIN 12 // Ultrasonic echo #define RFID_SS_PIN 5 // RFID SDA #define RFID_RST_PIN 0 // RFID Reset #define LED_GREEN 2 // Green LED (attendance OK) #define LED_RED 4 // Red LED (rejected) #define BUZZER_PIN 15 // Buzzer for audio feedback // ---- OLED Display ---- #define SCREEN_WIDTH 128 #define SCREEN_HEIGHT 64 Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1); // ---- RFID Reader ---- MFRC522 rfid(RFID_SS_PIN, RFID_RST_PIN); // ---- WiFi Credentials (Wokwi virtual WiFi) ---- const char* ssid = "Wokwi-GUEST"; const char* password = ""; // ---- Registered Users (simulate database) ---- struct User { String uid; String name; String id; }; User registeredUsers[] = { {"a1b2c3d4", "Rahim Ahmed", "STU-001"}, {"e5f6a7b8", "Fatima Khan", "STU-002"}, {"c9d0e1f2", "Karim Hossain", "STU-003"}, }; const int NUM_USERS = 3; // ---- Attendance Log ---- int attendanceCount = 0; // ============================================ void setup() { Serial.begin(115200); // Initialize pins pinMode(PIR_PIN, INPUT); pinMode(TRIG_PIN, OUTPUT); pinMode(ECHO_PIN, INPUT); pinMode(LED_GREEN, OUTPUT); pinMode(LED_RED, OUTPUT); pinMode(BUZZER_PIN, OUTPUT); // Initialize OLED if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) { Serial.println("OLED failed!"); } display.clearDisplay(); display.setTextSize(1); display.setTextColor(WHITE); display.setCursor(10, 10); display.println("Attendance System"); display.setCursor(10, 25); display.println("Initializing..."); display.display(); // Initialize SPI & RFID SPI.begin(); rfid.PCD_Init(); // Connect WiFi Serial.println("Connecting to WiFi..."); WiFi.begin(ssid, password); while(WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); } Serial.println("\nWiFi Connected!"); Serial.print("IP: "); Serial.println(WiFi.localIP()); // Ready screen showReadyScreen(); Serial.println("=== System Ready ==="); } // ============================================ void loop() { // Step 1: Check PIR motion int motionDetected = digitalRead(PIR_PIN); // Step 2: Read ultrasonic distance float distance = getDistance(); if (motionDetected == HIGH || distance < 50) { Serial.println("👤 Motion detected! Waiting for RFID..."); showMotionDetected(); // Step 3: Wait for RFID scan (5 second window) unsigned long startTime = millis(); while(millis() - startTime < 5000) { if(rfid.PICC_IsNewCardPresent() && rfid.PICC_ReadCardSerial()) { String cardUID = getCardUID(); Serial.print("Card UID: "); Serial.println(cardUID); // Step 4: Check if registered int userIndex = findUser(cardUID); if(userIndex >= 0) { // ✅ ATTENDANCE MARKED! attendanceCount++; markAttendance(registeredUsers[userIndex]); } else { // ❌ REJECTED rejectCard(cardUID); } rfid.PICC_HaltA(); rfid.PCD_StopCrypto1(); break; } delay(100); } } showReadyScreen(); delay(200); } // ---- Helper Functions ---- float getDistance() { digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2); digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10); digitalWrite(TRIG_PIN, LOW); long duration = pulseIn(ECHO_PIN, HIGH, 30000); return duration * 0.034 / 2; } String getCardUID() { String uid = ""; for(byte i = 0; i < rfid.uid.size; i++) { uid += String(rfid.uid.uidByte[i], HEX); } return uid; } int findUser(String uid) { for(int i = 0; i < NUM_USERS; i++) { if(registeredUsers[i].uid == uid) return i; } return -1; } void markAttendance(User user) { Serial.println("✅ ATTENDANCE MARKED: " + user.name); // Green LED + Buzzer beep digitalWrite(LED_GREEN, HIGH); tone(BUZZER_PIN, 1000, 200); // Show on OLED display.clearDisplay(); display.setTextSize(1); display.setCursor(0, 0); display.println("=== WELCOME ==="); display.setTextSize(2); display.setCursor(0, 16); display.println(user.name); display.setTextSize(1); display.setCursor(0, 40); display.println("ID: " + user.id); display.setCursor(0, 52); display.println("Attendance #" + String(attendanceCount)); display.display(); delay(2000); digitalWrite(LED_GREEN, LOW); } void rejectCard(String uid) { Serial.println("❌ REJECTED: Unknown card " + uid); // Red LED + Error buzzer digitalWrite(LED_RED, HIGH); tone(BUZZER_PIN, 300, 500); display.clearDisplay(); display.setTextSize(2); display.setCursor(10, 10); display.println("ACCESS"); display.println(" DENIED!"); display.setTextSize(1); display.setCursor(0, 50); display.println("Card: " + uid); display.display(); delay(2000); digitalWrite(LED_RED, LOW); } void showReadyScreen() { display.clearDisplay(); display.setTextSize(1); display.setCursor(5, 0); display.println("ATTENDANCE SYSTEM"); display.drawLine(0, 10, 128, 10, WHITE); display.setCursor(5, 18); display.println("Status: Ready"); display.setCursor(5, 30); display.println("WiFi: Connected"); display.setCursor(5, 42); display.println("Total: " + String(attendanceCount)); display.setCursor(5, 54); display.println("Scan card or move..."); display.display(); } void showMotionDetected() { display.clearDisplay(); display.setTextSize(1); display.setCursor(10, 5); display.println("MOTION DETECTED!"); display.setTextSize(2); display.setCursor(10, 25); display.println("SCAN"); display.println(" CARD"); display.display(); }
